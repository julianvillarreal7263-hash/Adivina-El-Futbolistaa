        function startGame() {
            // Shuffle questions to offer replayability
            currentQuestions = [...questionsDatabase].sort(() => Math.random() - 0.5);

        function loadQuestion() {
            answered = false;
            feedbackBox.classList.add('hidden');
            nextBtn.classList.add('hidden');

            const q = currentQuestions[currentQuestionIndex];

            // Update Progress Bar
            const progressPercent = ((currentQuestionIndex) / currentQuestions.length) * 100;
            progressBar.style.width = `${progressPercent}%`;

            // Question Counters and Info
            questionNumberEl.textContent = `PREGUNTA ${currentQuestionIndex + 1} / ${currentQuestions.length}`;
            difficultyBadgeEl.textContent = `Dificultad: ${q.difficulty}`;
            
            // Set difficulty color
            if (q.difficulty === 'Fácil') {
                difficultyBadgeEl.className = 'text-xs font-bold text-emerald-400 uppercase tracking-widest bg-emerald-950/40 px-3 py-1 rounded-full border border-emerald-500/30';
            } else if (q.difficulty === 'Media') {
                difficultyBadgeEl.className = 'text-xs font-bold text-yellow-400 uppercase tracking-widest bg-yellow-950/40 px-3 py-1 rounded-full border border-yellow-500/30';
            } else {
                difficultyBadgeEl.className = 'text-xs font-bold text-rose-400 uppercase tracking-widest bg-rose-950/40 px-3 py-1 rounded-full border border-rose-500/30';
            }

            questionTitleEl.textContent = q.question;

            // Render Clues
            cluesContainer.innerHTML = '';
            q.clues.forEach(clue => {
                const badge = document.createElement('span');
                badge.className = 'text-xs font-semibold bg-blue-900/60 text-cyan-300 border border-cyan-500/30 px-2.5 py-1 rounded-md flex items-center gap-1.5';
                badge.innerHTML = `<i class="fa-solid fa-lightbulb text-yellow-400"></i> ${clue}`;
                cluesContainer.appendChild(badge);
            });

            // Render Options (A, B, C, D)
            optionsContainer.innerHTML = '';
            q.options.forEach((opt, index) => {
                const btn = document.createElement('button');
                btn.className = 'option-btn text-left p-4 rounded-xl font-semibold text-sm md:text-base text-gray-100 flex items-center justify-between w-full';
                btn.innerHTML = `
                    <span>${opt}</span>
                    <i class="fa-regular fa-circle text-gray-500 opacity-60 status-icon"></i>
                `;
                btn.onclick = () => selectOption(index);
                optionsContainer.appendChild(btn);
            });
        }

        function selectOption(selectedIndex) {
            if (answered) return;
            answered = true;

            const q = currentQuestions[currentQuestionIndex];
            const optionButtons = optionsContainer.children;

            // Disable all option buttons
            Array.from(optionButtons).forEach(btn => {
                btn.disabled = true;
                btn.classList.add('cursor-not-allowed');
            });

            const isCorrect = selectedIndex === q.correctIndex;

            if (isCorrect) {
                // Award points
                score += 10;
                scoreDisplay.textContent = score;

                // Style chosen button as correct
                const selectedBtn = optionButtons[selectedIndex];
                selectedBtn.classList.add('correct-answer');
                selectedBtn.querySelector('.status-icon').className = 'fa-solid fa-circle-check text-emerald-400 text-lg';

                // Display Correct Feedback
                feedbackBox.className = 'rounded-xl p-4 mb-4 border bg-emerald-950/40 border-emerald-500/40 text-emerald-200 block animate-fadeIn';
                feedbackIcon.className = 'fa-solid fa-circle-check text-emerald-400 text-3xl';
                feedbackTitle.textContent = '¡ACIERTO GOLEADOR! (+10 PTS)';
                feedbackTitle.className = 'font-teko text-2xl font-bold tracking-wide text-emerald-400';
            } else {
                // Style chosen button as wrong
                const selectedBtn = optionButtons[selectedIndex];
                selectedBtn.classList.add('wrong-answer');
                selectedBtn.querySelector('.status-icon').className = 'fa-solid fa-circle-xmark text-rose-400 text-lg';

                // Highlight correct answer
                const correctBtn = optionButtons[q.correctIndex];
                correctBtn.classList.add('correct-answer');
                correctBtn.querySelector('.status-icon').className = 'fa-solid fa-circle-check text-emerald-400 text-lg';

                // Display Wrong Feedback
                feedbackBox.className = 'rounded-xl p-4 mb-4 border bg-rose-950/40 border-rose-500/40 text-rose-200 block animate-fadeIn';
                feedbackIcon.className = 'fa-solid fa-circle-xmark text-rose-400 text-3xl';
                feedbackTitle.textContent = '¡TARJETA AMARILLA! (RESPUESTA INCORRECTA)';
                feedbackTitle.className = 'font-teko text-2xl font-bold tracking-wide text-rose-400';
            }

            feedbackFact.textContent = q.fact;
            feedbackBox.classList.remove('hidden');

            // Show Next Button
            nextBtn.classList.remove('hidden');
        }

        function nextQuestion() {
            currentQuestionIndex++;

            if (currentQuestionIndex < currentQuestions.length) {
                loadQuestion();
            } else {
                showResults();
            }
        }
